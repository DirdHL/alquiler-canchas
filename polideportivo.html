Pasos para Crear la Tabla en Supabase (ignora esto xD)

  1. Entra a tu panel de base y crea un tabla.
  2. Ve a Configuración → Claves API y copia la clave anon | public de la pestaña Legacy. ⚠️ NO copies la clave secreta (service_role).
  3. Ve al Editor SQL en el menú lateral izquierdo.
  4. Haz clic en Nueva consulta, pega el siguiente SQL y presiona Ejecutar - By Rogger Dird: 
-- 1. Tabla de reservas
CREATE TABLE IF NOT EXISTS reservas (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  name TEXT NOT NULL,
  dni TEXT,
  medio TEXT,
  court TEXT NOT NULL,
  sport TEXT NOT NULL,
  date DATE NOT NULL,
  start_time TIME NOT NULL,
  end_time TIME NOT NULL,
  notes TEXT,
  pelota BOOLEAN DEFAULT false,
  chaleco BOOLEAN DEFAULT false,
  created_at TIMESTAMPTZ DEFAULT now()
);

-- Si ya tienes la tabla creada, ejecuta esto para agregar DNI, Medio, Pelota y Chaleco:
-- ALTER TABLE reservas ADD COLUMN IF NOT EXISTS dni TEXT;
-- ALTER TABLE reservas ADD COLUMN IF NOT EXISTS medio TEXT;
-- ALTER TABLE reservas ADD COLUMN IF NOT EXISTS pelota BOOLEAN DEFAULT false;
-- ALTER TABLE reservas ADD COLUMN IF NOT EXISTS chaleco BOOLEAN DEFAULT false;

ALTER TABLE reservas ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Acceso publico" ON reservas FOR ALL USING (true) WITH CHECK (true);

-- 2. Tabla de historial (Log de auditoria)
CREATE TABLE IF NOT EXISTS historial (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  action TEXT NOT NULL,         -- 'crear', 'editar', 'eliminar'
  user_name TEXT NOT NULL,      -- Nombre de quien modifico
  details TEXT NOT NULL,        -- Descripcion de la accion
  created_at TIMESTAMPTZ DEFAULT now()
);

ALTER TABLE historial ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Acceso publico" ON historial FOR ALL USING (true) WITH CHECK (true);

-- Habilitar acceso en tiempo real para ambas tablas
ALTER PUBLICATION supabase_realtime ADD TABLE reservas, historial;
